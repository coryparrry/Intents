    case first, second
    static let typeDisplayRepresentation: TypeDisplayRepresentation = "Qualification choice"

struct QualificationReference: AppEntity {
    static let typeDisplayRepresentation: TypeDisplayRepresentation = "Qualification reference"
    static let defaultQuery = QualificationReferenceQuery()
    var id: String
    @Property(title: "Label") var label: String
    init(id: String) { self.id = id; self.label = id }
    var displayRepresentation: DisplayRepresentation { .init(title: "\(id)") }
}
struct QualificationReferenceQuery: EntityStringQuery {
    func entities(matching string: String) async throws -> [QualificationReference] {
        ["canary-1", "canary-2"].filter { $0.contains(string) }.map { QualificationReference(id: $0) }
    }
    func entities(for identifiers: [String]) async throws -> [QualificationReference] {
        identifiers.filter { ["canary-1", "canary-2"].contains($0) }.map { QualificationReference(id: $0) }
    }
    func suggestedEntities() async throws -> [QualificationReference] { [.init(id: "canary-1"), .init(id: "canary-2")] }
}
struct QualificationEchoEntityIntent: AppIntent {
    static let title: LocalizedStringResource = "Qualification echo Entity"
    @Parameter(title: "Value") var value: QualificationReference
    func perform() async throws -> some IntentResult & ReturnsValue<String> { .result(value: value.id) }
}
